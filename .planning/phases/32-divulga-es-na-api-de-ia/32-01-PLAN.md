module AiSmoke
  KEY = Rails.application.credentials.dig(:api, :ai_key) || (ENV["AI_API_KEY"] ||= "ak_smoke_#{SecureRandom.hex(8)}")
  World = Struct.new(:client, :instance, :g1, :g2, :arte, keyword_init: true)

  @failures = []
  @session = ActionDispatch::Integration::Session.new(Rails.application).tap { |s| s.host! "localhost" }

  class << self
    attr_reader :failures

    # Tudo dentro de uma transacao que SEMPRE reverte (after_commit nunca dispara; nada persiste).
    def run
      ActiveRecord::Base.transaction do
        yield
        raise ActiveRecord::Rollback
      end
    end

    def jobs = ActiveJob::Base.queue_adapter.enqueued_jobs

    # Devolve [status, corpo_json_ou_nil]. auth: false omite o Bearer.
    def call(verb, path, body = nil, auth: true)
      headers = { "Content-Type" => "application/json", "Accept" => "application/json" }
      headers["Authorization"] = "Bearer #{KEY}" if auth
      @session.public_send(verb, path, params: body && body.to_json, headers: headers)
      parsed = begin
        JSON.parse(@session.response.body)
      rescue JSON::ParserError
        nil
      end
      [ @session.response.status, parsed ]
    end

    def check(label, condition)
      puts "#{condition ? 'ok  ' : 'FAIL'} #{label}"
      @failures << label unless condition
      condition
    end

    def finish!(marker)
      abort "FAILED (#{@failures.size}): #{@failures.join(' | ')}" if @failures.any?
      puts marker
    end

    def build_world!(tag = SecureRandom.hex(3))
      client   = Client.create!(name: "SMOKE #{tag}", password: "SmokePass123!", active: true)
      instance = client.create_whatsapp_instance!(instance_name: WhatsappInstance.evolution_name_for(client),

      g1 = instance.whatsapp_groups.create!(remote_jid: "smoke-#{tag}-1@g.us", subject: "Grupo Um", active: true, synced_at: Time.current)
      g2 = instance.whatsapp_groups.create!(remote_jid: "smoke-#{tag}-2@g.us", subject: "Grupo Dois", active: true, synced_at: Time.current)
      World.new(client: client, instance: instance, g1: g1, g2: g2, arte: build_arte!(client))
    end

    # no ActiveStorage do dev. save!(validate: false) pula so a regra "precisa de arquivo ou link" da Arte.

                   status: :approved, title: "Smoke", caption: "x" }

    # (sem callbacks/broadcast).
    def make_divulgacao!(world, at:, status: :agendada, groups: [ world.g1 ])

      groups.each { |g| d.divulgacao_grupos.build(whatsapp_group: g, group_name: g.display_name, remote_jid: g.remote_jid) }
      d.save!
      d.update_columns(status: Divulgacao.statuses.fetch(status.to_s)) unless status.to_s == "agendada"
      d
    end
  end
end

## Risks

- **POST nao idempotente (D-02, aceito):** nao ha chave de idempotencia nem unicidade (arte, cliente, horario);
  um retry apos timeout cria uma segunda divulgacao e cada grupo real recebe duas mensagens. Mitigacao
  procedural: o contrato (comentario de cabecalho do controller) manda consultar a listagem do cliente
  filtrada por `status=agendada` antes de repetir um POST que deu timeout; throttle de 60/min por key limita
  o estrago. Reconsiderar `confirm: true`/idempotency-key so se houver incidente (ver "Deferred Ideas" do
  32-CONTEXT).

  parseavel e a validacao existente responde "Informe a data e hora do envio." — aceito (D-08: sem validacao
  nova); o formato exigido (ISO 8601 com offset) e so contratual, documentado no cabecalho.
